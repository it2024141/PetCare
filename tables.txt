User 
userID INT PK 
username VARCHAR(50) UNIQUE NOT NULL 
password VARCHAR(255) NOT NULL 
email VARCHAR(100) UNIQUE NOT NULL 
phoneNumber VARCHAR(20) 
name VARCHAR(50) NOT NULL 
lastName VARCHAR(50) NOT NULL 
role VARCHAR(50) NOT NULL, δηλαδή Vet ή Οwner 

Availability 
availabilityID INT PK 
vetID INT FK->User(userID) NOT NULL 
availabilityDate DATE NOT NULL 
startTime TIME NOT NULL 
endTime TIME NOT NULL 
status VARCHAR(15) NOT NULL, δηλαδή PENDING, CONFIRMED, CANCELLED, COMPLETED 

Vaccination 
vaccineID INT PK 
petID INT FK->Pet(petID) NOT NULL 
vetID INT FK->User(userID) NOT NULL

dateAdministered DATE NOT NULL 
nextDueDate DATE 
notes TEXT 

appointmentID INT PK
petID INT FK->Pet(petID) NOT NULL
vetID INT FK->User(userID) NOT NULL
appointmentDate DATE NOT NULL
startTime TIME NOT NULL
endTime TIME NOT NULL

comments TEXT
